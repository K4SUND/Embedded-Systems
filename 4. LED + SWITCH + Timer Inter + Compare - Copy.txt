

#include <avr/io.h>
#include <avr/interrupt.h>
#define F_CPU 16000000L

void configureTimerINT(void)
{
	//prescalar (256)
	TCCR1B |= (1<<CS12);
	TCCR1B &= ~((1<<CS10)|(1<<CS11));
	
	//mode (Compare CTC)
	TCCR1A &= ~((1<<WGM11)|(1<<WGM10));
	TCCR1B |= (1<<WGM12);
	TCCR1B &= ~(1<<WGM13);
	
	//compare interrupt enable (A)
	TIMSK1 |= (1<<OCIE1A);
	
	//compare value set
	OCR1AH = 0xF4;
	OCR1AL = 0x23;
	
	sei();
	
	
	
}

volatile int toggle = 0;


int main(void)
{
    
   //Data direction
   DDRB |= (1<<PB5);
   
   
 
   
   configureTimerINT();
   
   
   
    while (1) 
    {
		
		
		
    }
}

ISR(TIMER1_COMPA_vect)
{
	if(toggle)
	{
		PORTB &= ~(1<<PB5);
		
	}
	else{
		
		PORTB |= (1<<PB5);
	}
	
	toggle = 1 - toggle  ;

}
